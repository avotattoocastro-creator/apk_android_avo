cmake_minimum_required(VERSION 3.16)
project(avo_equine_intelligence VERSION 1.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

option(AVO_EQ_BUILD_TOOLS "Build CLI, fixtures and tests (OFF inside the Android app build)" ON)
option(AVO_EQ_SANITIZE "Build with AddressSanitizer + UndefinedBehaviorSanitizer" OFF)
if(AVO_EQ_SANITIZE AND NOT MSVC)
  add_compile_options(-fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
if(NOT MSVC)
  add_compile_options(-Wall -Wextra -Wno-unused-parameter)
endif()

# The RAW session format is read with the same avo_core code that the chest uses to write it.
set(AVO_CORE_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../firmware/back/V3_8_0/lib/avo_core/src CACHE PATH "avo_core sources")
if(MSVC)
  add_compile_definitions(_USE_MATH_DEFINES _CRT_SECURE_NO_WARNINGS)
  include_directories(${CMAKE_CURRENT_SOURCE_DIR}/compat/msvc)
endif()
file(GLOB AVO_CORE_SRC ${AVO_CORE_DIR}/*.cpp)

file(GLOB_RECURSE EQ_SRC ${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp)
# PPK (motor 2.0.0-dev32): RTKLIB demo5 in C (src/ppk/rtklib, BSD 2-clause; see its README_AVO.md). Third-party code:
# compiled without warnings and without sanitizers (the AVO code around it keeps both).
file(GLOB RTKLIB_SRC ${CMAKE_CURRENT_SOURCE_DIR}/src/ppk/rtklib/*.c ${CMAKE_CURRENT_SOURCE_DIR}/src/ppk/rtklib/rcv/*.c)
if(MSVC)
  set_source_files_properties(${RTKLIB_SRC} PROPERTIES COMPILE_OPTIONS "/w")
else()
  set_source_files_properties(${RTKLIB_SRC} PROPERTIES COMPILE_OPTIONS "-w;-fno-sanitize=all")
endif()
list(APPEND EQ_SRC ${RTKLIB_SRC})
find_package(Threads)
add_library(avo_eq STATIC ${EQ_SRC} ${AVO_CORE_SRC})
target_compile_definitions(avo_eq PUBLIC AVO_EQ_STATIC)
target_include_directories(avo_eq PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include PRIVATE ${AVO_CORE_DIR})
if(Threads_FOUND)
  target_link_libraries(avo_eq PUBLIC Threads::Threads)
endif()
if(NOT MSVC AND NOT ANDROID AND NOT APPLE)
  target_link_libraries(avo_eq PUBLIC m)
endif()
if(WIN32)
  target_link_libraries(avo_eq PUBLIC winmm)
endif()

# Shared library with the C API (Android JNI, Windows P/Invoke, Python ctypes)
add_library(avo_eq_shared SHARED ${EQ_SRC} ${AVO_CORE_SRC})
target_include_directories(avo_eq_shared PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include PRIVATE ${AVO_CORE_DIR})
target_compile_definitions(avo_eq_shared PRIVATE AVO_EQ_BUILD_DLL)
if(Threads_FOUND)
  target_link_libraries(avo_eq_shared PRIVATE Threads::Threads)
endif()
if(NOT MSVC AND NOT ANDROID AND NOT APPLE)
  target_link_libraries(avo_eq_shared PRIVATE m)
endif()
if(WIN32)
  target_link_libraries(avo_eq_shared PRIVATE winmm)
endif()
set_target_properties(avo_eq_shared PROPERTIES OUTPUT_NAME avo_equine)

if(AVO_EQ_BUILD_TOOLS)
add_executable(avo_eq_cli tools/session-inspector/avo_eq_cli.cpp)
target_link_libraries(avo_eq_cli avo_eq)
add_executable(avo_training_cli tools/training/avo_training_cli.cpp)
target_link_libraries(avo_training_cli avo_eq)

enable_testing()
file(GLOB EQ_TEST_SRC ${CMAKE_CURRENT_SOURCE_DIR}/tests/*.cpp)
add_executable(avo_eq_tests ${EQ_TEST_SRC})
target_include_directories(avo_eq_tests PRIVATE ${AVO_CORE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/tests)
target_link_libraries(avo_eq_tests avo_eq)
add_test(NAME avo_eq_tests COMMAND avo_eq_tests)
add_executable(avo_eq_fixtures tools/fixtures/make_fixtures.cpp)
target_include_directories(avo_eq_fixtures PRIVATE ${AVO_CORE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/tests)
target_link_libraries(avo_eq_fixtures avo_eq)
endif()
